cmake_minimum_required(VERSION 3.20)
project(licensebox_cpp VERSION 0.2.0 LANGUAGES C CXX)
include(c-version.cmake)
include(cmake/NativeCandidate.cmake)
include(CTest)
include(GNUInstallDirs)
if(NOT TARGET licensebox::c AND NOT LICENSEBOX_C_CANDIDATE)
  find_package(licensebox 0.4 CONFIG QUIET)
endif()
if(NOT TARGET licensebox::c)
  include(FetchContent)
  set(LICENSEBOX_C_SOURCE_DIR "" CACHE PATH "Local C SDK checkout for development")
  if(LICENSEBOX_C_SOURCE_DIR AND NOT LICENSEBOX_C_CANDIDATE)
    FetchContent_Declare(licensebox_c SOURCE_DIR "${LICENSEBOX_C_SOURCE_DIR}")
  elseif(LICENSEBOX_C_CANDIDATE)
    FetchContent_Declare(licensebox_c GIT_REPOSITORY https://github.com/licensebox/c.git GIT_TAG ${LICENSEBOX_C_VERSION} GIT_SHALLOW FALSE)
  else()
    FetchContent_Declare(licensebox_c GIT_REPOSITORY https://github.com/licensebox/c.git GIT_TAG ${LICENSEBOX_C_VERSION} GIT_SHALLOW TRUE)
  endif()
  FetchContent_MakeAvailable(licensebox_c)
endif()
add_library(licensebox_cpp INTERFACE)
add_library(licensebox::cpp ALIAS licensebox_cpp)
set_target_properties(licensebox_cpp PROPERTIES EXPORT_NAME cpp)
target_compile_features(licensebox_cpp INTERFACE cxx_std_17)
target_include_directories(licensebox_cpp INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(licensebox_cpp INTERFACE licensebox::c)
if(BUILD_TESTING)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
  add_executable(licensebox-cpp-adapter tests/adapter.cpp)
  target_link_libraries(licensebox-cpp-adapter PRIVATE licensebox::cpp PkgConfig::JSONC)
  target_compile_options(licensebox-cpp-adapter PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(DIRECTORY include/licensebox DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS licensebox_cpp EXPORT licenseboxCppTargets)
install(EXPORT licenseboxCppTargets NAMESPACE licensebox:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/licensebox-cpp)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/licensebox-cppConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/licensebox-cppConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/licensebox-cpp)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/licensebox-cppConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/licensebox-cpp)
